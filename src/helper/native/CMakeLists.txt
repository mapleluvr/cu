cmake_minimum_required(VERSION 3.20)
project(cu_capture LANGUAGES CXX)
add_executable(cu-capture capture.cpp)
option(CU_BUILD_TESTS "Build desktop-free native core checks" OFF)
if(CU_BUILD_TESTS)
  add_executable(cu-capture-core ../../../test/native/capture-core.cpp)
endif()
set(cu_targets cu-capture)
if(CU_BUILD_TESTS)
  list(APPEND cu_targets cu-capture-core)
endif()
foreach(target IN LISTS cu_targets)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(${target} PRIVATE user32 gdi32 gdiplus ole32 bcrypt wtsapi32)
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endforeach()
